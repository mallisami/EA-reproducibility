4. VERIFICATION PROBES (in $NS; logged in audit + webhook logs, unmatched to any trial)
--------------------------------------------------------------------------------------------------
  date -u +%FT%T.%NZ | tee $R/PROBES_START.txt
  # 4.1 valid fixture ConfigMaps (the runner's reset later patches these same objects)
  for x in a b c d e f; do k -n $NS create configmap evidence-$x --from-literal=value=true; done
  k -n $NS create configmap response-scope --from-literal=allowed=true
  k -n $NS create configmap application-policy --from-literal=allowed=true
  k -n $NS create configmap e38-epoch --from-literal=sequence=0
  k -n $NS label configmap --all e38-managed=true
  # 4.2 dummy pods: two with the webhook arm label, one without
  for p in probe-deny probe-allow probe-reset; do
    k -n $NS run $p --image=python:3.12-slim --image-pull-policy=Never --restart=Never \
      --labels=e38-managed=true,e38v2-webhook=enforce -- sleep 3600; done
  k -n $NS run probe-nolabel --image=python:3.12-slim --image-pull-policy=Never --restart=Never \
      --labels=e38-managed=true -- sleep 3600
  k -n $NS wait --for=condition=Ready pod --all --timeout=120s
  # 4.3 INVALID state -> webhook must DENY the admin's DELETE
  k -n $NS patch configmap application-policy --type=merge -p '{"data":{"allowed":"false"}}'
  k -n $NS delete pod probe-deny --wait=false ; echo "exit=$?"
      # EXPECT non-zero exit, message: admission webhook "delete-revalidation..." denied ... app_policy_not_allowed
  k -n $NS get pod probe-deny                 # EXPECT still Running, no deletionTimestamp
  # 4.4 INVALID state, pod WITHOUT arm label -> not intercepted (A0-A2 behaviour)
  k -n $NS delete pod probe-nolabel --wait=false ; echo "exit=$?"      # EXPECT 0
  # 4.5 INVALID state, reset identity -> exempt by matchConditions
  k -n $NS delete pod probe-reset --wait=false \
      --as=system:serviceaccount:e38v2-system:e38v2-reset ; echo "exit=$?"   # EXPECT 0
  # 4.6 VALID state -> ALLOW
  k -n $NS patch configmap application-policy --type=merge -p '{"data":{"allowed":"true"}}'
  k -n $NS delete pod probe-allow --wait=true ; echo "exit=$?"         # EXPECT 0, pod gone (kubelet final delete exempt)
  k -n $NS delete pod probe-deny  --wait=true ; echo "exit=$?"         # EXPECT 0
  # 4.7 each other invalid predicate denies (optional but recommended)
  for spec in 'response-scope allowed' 'evidence-a value'; do set -- $spec
    k -n $NS run probe-x --image=python:3.12-slim --image-pull-policy=Never --restart=Never \
      --labels=e38-managed=true,e38v2-webhook=enforce -- sleep 3600
    k -n $NS wait --for=condition=Ready pod/probe-x --timeout=60s
    k -n $NS patch configmap $1 --type=merge -p "{\"data\":{\"$2\":\"false\"}}"
    [ $1 = evidence-a ] && k -n $NS patch configmap evidence-c --type=merge -p '{"data":{"value":"false"}}' \
                        && k -n $NS patch configmap evidence-e --type=merge -p '{"data":{"value":"false"}}'
    k -n $NS delete pod probe-x --wait=false; echo "EXPECT DENY exit=$?"
    for c in evidence-a evidence-c evidence-e; do k -n $NS patch configmap $c --type=merge -p '{"data":{"value":"true"}}'; done
    k -n $NS patch configmap response-scope --type=merge -p '{"data":{"allowed":"true"}}'
    k -n $NS delete pod probe-x --wait=true; echo "EXPECT ALLOW exit=$?"
  done
  # 4.8 logs
  tail -n 20 $E/results/_webhook_host/$RUN_ID/webhook_log.jsonl
      # EXPECT: schema e38v2.webhook.v1; deny lines for probe-deny/probe-x; allow lines; 8 reads each
      #         with numeric resource_version; NO lines for probe-nolabel / probe-reset
  sudo python3 - "$E/results/_audit_host/audit.log" "$NS" <<'PY'
import json,sys
ev=[json.loads(l) for l in open(sys.argv[1])]
mine=[e for e in ev if e.get("objectRef",{}).get("namespace")==sys.argv[2]]
print("events",len(ev),"in test ns",len(mine))
for e in mine[-12:]:
    ro=e.get("responseObject") or {}
    print(e["stage"],e["verb"],e["objectRef"].get("resource"),e["objectRef"].get("name"),
          e["responseStatus"].get("code"),e["auditID"],(ro.get("metadata") or {}).get("resourceVersion"),
          e["requestReceivedTimestamp"],e["stageTimestamp"])
assert all(e["stage"]!="RequestReceived" for e in ev)
assert any(e["verb"]=="patch" and e["objectRef"]["resource"]=="configmaps" and e.get("responseObject") for e in mine)
assert any(e["verb"]=="delete" and e["objectRef"]["resource"]=="pods" for e in mine)
print("AUDIT OK")
PY
  # the denied DELETE must appear in audit with code 403 and the webhook message in responseStatus.
  k -n $NS delete pod --all --as=system:serviceaccount:e38v2-system:e38v2-reset --wait=true
